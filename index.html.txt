<script>
    // ---- ЗАМЕНИТЕ ЭТУ ССЫЛКУ ПОСЛЕ НАСТРОЙКИ GOOGLE ТАБЛИЦЫ ----
    const GOOGLE_SCRIPT_URL = 'ВАША_ССЫЛКА_НА_GOOGLE_APPS_SCRIPT';

    // Состояние ответов
    let answers = {
        q1: null, q2: null, q3: null,
        q4: null, q5: null,
        q6: null, q7: null, q8: null
    };

    // Активация кнопок выбора
    const questions = document.querySelectorAll('.question');
    questions.forEach(questionDiv => {
        const btns = questionDiv.querySelectorAll('.option-btn');
        btns.forEach(btn => {
            btn.addEventListener('click', (e) => {
                const value = btn.getAttribute('data-value');
                const qName = questionDiv.getAttribute('data-q');
                
                // Убираем активный класс у всех кнопок в этом вопросе
                btns.forEach(b => b.classList.remove('active'));
                btn.classList.add('active');
                
                // Сохраняем ответ
                if (answers.hasOwnProperty(qName)) {
                    answers[qName] = value;
                }
                
                // Скрываем ошибку, если была
                const errorSpan = questionDiv.querySelector('.error-marker');
                if (errorSpan) errorSpan.style.display = 'none';
            });
        });
    });

    // Проверка заполнения всех вопросов
    function isAllAnswered() {
        for (let key in answers) {
            if (answers[key] === null) return false;
        }
        return true;
    }

    // Показать ошибки по незаполненным вопросам
    function showMissingErrors() {
        let allOk = true;
        questions.forEach(questionDiv => {
            const qName = questionDiv.getAttribute('data-q');
            const errorSpan = questionDiv.querySelector('.error-marker');
            if (answers[qName] === null) {
                if (errorSpan) errorSpan.style.display = 'block';
                allOk = false;
            } else {
                if (errorSpan) errorSpan.style.display = 'none';
            }
        });
        return allOk;
    }

    // Отправка данных в Google Таблицу
    async function sendToGoogleSheets(data) {
        if (!GOOGLE_SCRIPT_URL || GOOGLE_SCRIPT_URL === 'https://script.google.com/macros/s/AKfycbwDcs8U5p3t5g-K6gEUPJ7xXcyVUuYb8h22uJtHMq1dfJBLnSJPMdZZ7kHzAakrD2w9QA/exec') {
            console.warn('Google Script URL не настроен');
            return { success: false, message: 'Ссылка на Google Таблицу не настроена. Проверьте инструкцию.' };
        }
        
        try {
            const response = await fetch(GOOGLE_SCRIPT_URL, {
                method: 'POST',
                mode: 'no-cors', 
                headers: {
                    'Content-Type': 'application/json',
                },
                body: JSON.stringify(data)
            });
            // из-за no-cors нельзя прочитать ответ, но запрос уйдёт
            return { success: true, message: 'Спасибо! Ваши ответы сохранены.' };
        } catch (error) {
            console.error('Ошибка отправки:', error);
            return { success: false, message: 'Ошибка соединения. Попробуйте ещё раз.' };
        }
    }

    // Обработка отправки
    document.getElementById('submitBtn').addEventListener('click', async () => {
        const allFilled = showMissingErrors();
        if (!allFilled) {
            const msgDiv = document.getElementById('formMessage');
            msgDiv.className = 'message error';
            msgDiv.style.display = 'block';
            msgDiv.innerText = 'Пожалуйста, ответьте на все вопросы перед отправкой.';
            return;
        }

        // Добавляем метаданные: дата и время
        const payload = {
            timestamp: new Date().toLocaleString('ru-RU', { timeZone: 'Europe/Moscow' }),
            ...answers
        };

        // Отключаем кнопку, чтобы не дублировали
        const submitBtn = document.getElementById('submitBtn');
        submitBtn.disabled = true;
        submitBtn.textContent = 'Отправка...';

        const result = await sendToGoogleSheets(payload);
        
        const msgDiv = document.getElementById('formMessage');
        msgDiv.style.display = 'block';
        if (result.success) {
            msgDiv.className = 'message success';
            msgDiv.innerText = result.message;
            // Не очищаем форму, чтобы человек мог видеть свои ответы.
            // Но можно оставить активные выбранные варианты.
        } else {
            msgDiv.className = 'message error';
            msgDiv.innerText = result.message || 'Ошибка при отправке. Проверьте соединение.';
            submitBtn.disabled = false;
            submitBtn.textContent = 'Отправить ответы';
        }
    });
</script>
